/**
 * Definition for read4()
 * read4 = function(buf4: string[]): number {
 *     ...
 * };
 */
 var solution = function(read4: any) {
    const iter = iterateWithRetry(flatStream(read4, 4));

    return function(buf: string[], n: number): number {
        let len = 0;
        for (let value of iter) {
            if (len === n) {
                iter.retry();
                return n;
            }
            buf[len++] = value;
        }

        return len;
    }
 };

/*** ---- Reusable utilities that could be added to libraries. ----- ***/

type RetryableIterableIterator<T> = IterableIterator<T> & { retry: () => void };

/**
 * flatStream flattens a batched stream into a continuous sequence of data. It repeatedly
 * invokes the `read` callback to retrieve chunks of data from the source stream, then
 * yields each element individually. 
 *
 * @param read - A function that populates the provided buffer and returns the number of elements read.
 * @param max - The maximum capacity of the buffer required by read.
 * @yields The next individual element from the stream.
 */
function* flatStream(read, max): Generator<string, undefined, undefined> {
    let len;
    const mem = new Array<string>(max);
    
    while (len = read(mem)) 
        for (let i = 0; i < len; i++)
            yield mem[i];
}

/**
 * Wraps an iterator to add a `retry` capability, allowing the consumer to re-process 
 * the previously yielded value in the next iteration.
 *
 * @template T - The type of the value yielded by the iterator.
 * @param iterator - The source iterator to wrap.
 * @returns An object that implements the Iterator protocol and provides a `retry()`
 * method to backtrack one step.
 */
function iterateWithRetry<T>(iterator: Iterator<T>): RetryableIterableIterator<T> {
    let prev = null;
    let current = iterator.next();
    let next = null;
    
    return {
        next() {
            [prev, current, next] = [current, next || iterator.next(), null];
            return prev;
        },

        retry() {
            if (prev) [current, next] = [prev, current];
        },

        [Symbol.iterator]() {
            return this;
        }
    };
}
