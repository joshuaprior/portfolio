type Maze = number[][];
type Point = [number, number];

class Direction {
    #coordIndex: 0 | 1;
    #step: -1 | 1;
    
    constructor(coordIndex, step) {
        this.#coordIndex = coordIndex;
        this.#step = step;
    }

    // Roll a ball in the provided maze starting from the provided
    // point, then return the final position of the ball.
    roll(maze: Maze, start: Point): Point {
        const p: Point = [...start];
        let row, col;

        do {
            p[this.#coordIndex] += this.#step;
            [row, col] = p;
        } while (maze[row]?.[col] === 0);
        
        p[this.#coordIndex] -= this.#step;
        return p;
    }
}

function hasPath(maze: Maze, start: Point, goal: Point): boolean {
    const [goalRow, goalCol] = goal;

    const directions = [
        new Direction( 0, -1), // up
        new Direction( 0,  1), // down
        new Direction( 1, -1), // left
        new Direction( 1,  1), // right
    ];

    const visited = new Array(maze.length)
        .fill(null)
        .map(() => new Array(maze[0].length).fill(false));

    function hasPathFrom(p: Point): boolean {
        const [row, col] = p;
        if (visited[row][col]) return false;
        visited[row][col] = true;

        if (row === goalRow && col === goalCol) return true;

        return directions.some(dir => {
            const dest = dir.roll(maze, p);
            const [destRow, destCol] = dest;
            if (destRow === row && destCol === col) return false;
            return hasPathFrom(dest);
        });
    }

    return hasPathFrom(start);    
};
